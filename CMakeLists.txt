cmake_minimum_required(VERSION 3.16)
project(M1136MacDriver VERSION 0.1.0 LANGUAGES C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
option(M1136_SANITIZE "Enable ASan and UBSan" OFF)
option(M1136_BUILD_FILTER "Build the system CUPS filter" ON)
if(M1136_SANITIZE)
  add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer)
  add_link_options(-fsanitize=address,undefined)
endif()
add_compile_options(-fstack-protector-strong -ffunction-sections -fdata-sections)
if(APPLE)
  add_link_options(-Wl,-dead_strip)
else()
  add_link_options(-Wl,--gc-sections)
endif()
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/vendor/jbigkit/SOURCE.json" VENDOR_MANIFEST)
# Python is only a build/test dependency, never a runtime driver dependency.
find_package(Python3 REQUIRED COMPONENTS Interpreter)
execute_process(COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/scripts/verify-vendor.py
                RESULT_VARIABLE VENDOR_RESULT)
if(NOT VENDOR_RESULT EQUAL 0)
  message(FATAL_ERROR "Vendor integrity check failed")
endif()
add_library(jbig STATIC vendor/jbigkit/jbig.c vendor/jbigkit/jbig_ar.c)
target_include_directories(jbig PUBLIC vendor/jbigkit)
add_library(m1136 STATIC src/page.c src/xqx.c src/codec.c)
target_include_directories(m1136 PUBLIC include PRIVATE src)
target_compile_options(m1136 PRIVATE -Wall -Wextra -Wpedantic -Werror)
target_link_libraries(m1136 PRIVATE jbig)
enable_testing()
add_executable(jbig-selftest vendor/jbigkit/tstcodec.c)
target_link_libraries(jbig-selftest PRIVATE jbig)
add_test(NAME jbig-conformance COMMAND jbig-selftest)
add_executable(core-test tests/core.c)
target_link_libraries(core-test PRIVATE m1136 jbig)
target_compile_options(core-test PRIVATE -Wall -Wextra -Wpedantic -Werror)
add_test(NAME core COMMAND core-test)
