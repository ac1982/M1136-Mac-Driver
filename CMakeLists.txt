cmake_minimum_required(VERSION 3.16)
file(STRINGS "${CMAKE_CURRENT_SOURCE_DIR}/VERSION" M1136_VERSION LIMIT_COUNT 1)
project(M1136MacDriver VERSION ${M1136_VERSION} LANGUAGES C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
option(M1136_SANITIZE "Enable ASan and UBSan" OFF)
option(M1136_BUILD_FILTER "Build the system CUPS filter" ON)
if(M1136_SANITIZE)
  add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer)
  add_link_options(-fsanitize=address,undefined)
endif()
add_compile_options(-fstack-protector-strong -ffunction-sections -fdata-sections)
if(APPLE)
  add_link_options(-Wl,-dead_strip)
else()
  add_link_options(-Wl,--gc-sections)
endif()
# Python is only a build/test dependency, never a runtime driver dependency.
find_package(Python3 REQUIRED COMPONENTS Interpreter)
execute_process(COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/scripts/verify-vendor.py
                RESULT_VARIABLE VENDOR_RESULT)
if(NOT VENDOR_RESULT EQUAL 0)
  message(FATAL_ERROR "Vendor integrity check failed")
endif()
add_library(jbig STATIC vendor/jbigkit/jbig.c vendor/jbigkit/jbig_ar.c)
target_include_directories(jbig PUBLIC vendor/jbigkit)
target_compile_options(jbig PRIVATE -UNDEBUG)
add_library(m1136 STATIC src/page.c src/xqx.c src/codec.c)
target_include_directories(m1136 PUBLIC include PRIVATE src)
target_compile_options(m1136 PRIVATE -Wall -Wextra -Wpedantic -Werror)
target_link_libraries(m1136 PRIVATE jbig)
enable_testing()
# Upstream selftest leaves two arithmetic-coder allocations unfreed at exit.
# Add cleanup only to a generated test copy; vendored library stays byte-identical.
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/vendor/jbigkit/tstcodec.c" JBIG_TEST_SOURCE)
string(REPLACE "  return problems != 0;" "  free(se);\n  free(sd);\n  return problems != 0;"
       JBIG_TEST_SOURCE "${JBIG_TEST_SOURCE}")
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/jbig-selftest.c" "${JBIG_TEST_SOURCE}")
add_executable(jbig-selftest "${CMAKE_CURRENT_BINARY_DIR}/jbig-selftest.c")
target_link_libraries(jbig-selftest PRIVATE jbig)
add_test(NAME jbig-conformance COMMAND jbig-selftest)
add_executable(core-test tests/core.c)
target_link_libraries(core-test PRIVATE m1136 jbig)
target_compile_options(core-test PRIVATE -Wall -Wextra -Wpedantic -Werror)
add_test(NAME core COMMAND core-test)
if(M1136_BUILD_FILTER)
  find_package(Cups REQUIRED)
  add_executable(rastertom1136 src/filter.c src/raster_input.c src/io.c src/duplex.c)
  target_include_directories(rastertom1136 PRIVATE ${CUPS_INCLUDE_DIR})
  target_link_libraries(rastertom1136 PRIVATE m1136 ${CUPS_LIBRARIES})
  target_compile_options(rastertom1136 PRIVATE -Wall -Wextra -Wpedantic -Werror -Wno-deprecated-declarations)
endif()
if(M1136_BUILD_FILTER)
  add_executable(raster-fixture tests/raster_fixture.c)
  target_include_directories(raster-fixture PRIVATE ${CUPS_INCLUDE_DIR})
  target_link_libraries(raster-fixture PRIVATE ${CUPS_LIBRARIES})
  target_compile_options(raster-fixture PRIVATE -Wno-deprecated-declarations)
  add_test(NAME cups-filter COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/filter.py
           $<TARGET_FILE:rastertom1136> $<TARGET_FILE:raster-fixture>)
  set_tests_properties(cups-filter PROPERTIES TIMEOUT 180)
endif()

add_executable(io-test tests/io.c)
target_include_directories(io-test PRIVATE src)
target_compile_options(io-test PRIVATE -Wall -Wextra -Wpedantic -Werror)
add_test(NAME cancellation-races COMMAND io-test)
set_tests_properties(cancellation-races PROPERTIES TIMEOUT 5)
add_test(NAME build-provenance COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/provenance.py)
add_test(NAME release-version COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/release_version.py)
add_test(NAME release-assets COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/release_assets.py)

if(M1136_BUILD_FILTER)
  add_executable(jbig-decode-test tests/decode.c)
  target_link_libraries(jbig-decode-test PRIVATE jbig)
  add_test(NAME manual-duplex COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/duplex.py
           $<TARGET_FILE:rastertom1136> $<TARGET_FILE:raster-fixture> $<TARGET_FILE:jbig-decode-test>)
  set_tests_properties(manual-duplex PROPERTIES TIMEOUT 90)
endif()

if(M1136_BUILD_FILTER)
  add_executable(duplex-boundaries-test tests/duplex_boundaries.c src/io.c)
  target_include_directories(duplex-boundaries-test PRIVATE src ${CUPS_INCLUDE_DIR})
  target_link_libraries(duplex-boundaries-test PRIVATE m1136 ${CUPS_LIBRARIES})
  target_compile_options(duplex-boundaries-test PRIVATE -Wall -Wextra -Wpedantic -Werror -Wno-deprecated-declarations)
  add_test(NAME duplex-boundaries COMMAND duplex-boundaries-test)
  set_tests_properties(duplex-boundaries PROPERTIES TIMEOUT 10)
endif()
